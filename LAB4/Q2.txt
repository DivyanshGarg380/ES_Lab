/*
    Write an assembly language program to convert a 2-digit BCD number in to its
    equivalent hexadecimal number.  
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =BCD
    LDRB R1, [R0]

    ; Extract tens digit
    AND R2, R1, #0xF0
    MOV R2, R2, LSR #4

    ; Multiply tens digit by 10
    MOV R3, R2, LSL #3      ; R3 = R2 * 8
    ADD R2, R3, R2, LSL #1  ; R2 = R3 + R2 * 2 = R2 * 10

    ; Get units digit now
    AND R3, R1, #0x0F
    ADD R2, R2, R3

    LDR R0, =RESULT
    STR R2, [R0]

STOP
    B STOP

BCD DCD 0x25
    AREA data, DATA, READWRITE

RESULT DCD 0
    END
