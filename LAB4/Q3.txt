/*
    Write an assembly language program to convert a 2-digit hex number into its
    equivalent BCD number.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =Hex
    LDRB R1, [R0]
    MOV R2, #0              ; Tens digit 

LOOP
    CMP R1, #10
    BLO DONE

    SUB R1, R1, #10
    ADD R2, R2, #1
    B LOOP

DONE    
    ; R2 has the tens, R1 the units
    MOV R3, R2, LSL #4      ; Move to tens digit
    ADD R3, R3, R1

    LDR R0, =RESULT 
    STR R3, [R0]

STOP
    B STOP

Hex DCD 0x25
    AREA data, DATA, READWRITE

RESULT DCD 0
    END
