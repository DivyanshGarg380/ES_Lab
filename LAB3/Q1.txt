/*
    Write an Assembly code to add two 32-bit numbers
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =VAL1           ; Address of first number
    LDR R1, [R0]            ; actual value of VAL1

    LDR R0, =VAL2           ; Address of second number
    LDR R3, [R0]            ; actual value of VAL2

    ADDS R6, R1, R3         

    LDR R2, =RES            ; Address of result
    STR R6, [R2]            ; store result

STOP
    B STOP

VAL1 DCD 0x12345678

VAL2 DCD 0XABCDEF12
    AREA DATASEG, DATA, READWRITE

RES
    DCD 0
    END
