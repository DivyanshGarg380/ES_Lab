/*
    Write a program to multiply two 32 bit numbers using repetitive addition
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =VAL1
    LDR R0, [R0]            ; Number 1

    LDR R1, =VAL2
    LDR R1, [R1]            ; Number 2

    MOV R2, #0              ; Carry
    MOV R3, #0              ; Sum

LOOP
    ADDS R3, R3, R0         ; Sum = Sum + R0
    BCC NOCARRY             ; if no Carry

    ADD R2, R2, #1          ; else increment carry

NOCARRY
    SUB R1, R1, #1          ; R1--
    CMP R1, #0
    BNE LOOP

    LDR R4, =RESULT
    STR R3, [R4]            ; Store lower 32 bits
    STR R2, [R4, #4]        ; Store upper part

STOP
    B STOP

VALUE1 DCD 5
VALUE2 DCD 4
    AREA DATASEG, DATA, READWRITE

RESULT DCD 0, 0
    END
