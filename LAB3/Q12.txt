/*
    Write a program for BCD multiplication
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =VAL1
    LDR R1, [R0]

    LDR R0, =VAL2
    LDR R2, [R0]

    MOV R3, #0

LOOP
    ADD R3, R3, R1
    SUB R2, R2, #1

    CMP R2, #0
    BNE LOOP

    LDR R0, =RESULT
    STR R3, [R0]

STOP
    B STOP

VAL1 DCD 0x00000012
VAL2 DCD 0x00000003
    AREA DATASEG, DATA, READWRITE

RESULT DCD 0
    END
