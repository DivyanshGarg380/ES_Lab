/*
    Write an assembly language program to find GCD of two numbers
    Hint:
    while(a!=b)
    {
        if(a>b)
            a=a-b;
        else
            b=b-a;
    } return (a);
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =VAL1
    LDR R1, [R0]            ; a
    LDR R0, =VAL2
    LDR R2, [R0]            ; b 

LOOP
    CMP R1, R2
    BEQ DONE                ; GCD found

    SUBCS R1, R1, R2        ; if a > b then a = a - b 
    SUBCC R2, R2, R1        ; if a < b then b = b - a
    B LOOP

DONE
    MOV R3, R1
    LDR R0, =RESULT
    STR R3, [R0]

STOP
    B STOP

VAL1 DCD 48
VAL2 DCD 18
    AREA DAASEG, DATA, READWRITE

RESULT DCD 0
    END
