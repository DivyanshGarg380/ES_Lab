/*
    Write an Assembly code to divide a 32-bit number by a 16-bit number
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    MOV R2, #0              ; store quotient in this
    LDR R0, =VAL1           ; Address of dividend
    LDR R1, [R0]

    LDR R0, =VAL2           ; Address of divisor
    LDR R3, [R0]

UP
    SUB R1, R3              ; R1 = R1 - R3
    ADD R2, #1              ; Increment quotient

    CMP R1, R3
    BCS UP                  ; Loop back if R1 >= R3

    LDR R6, =RES
    STR R2, [R6, #4]        ; Store quotient
    STR R1, [R6]            ; Store remainder

STOP
    B STOP

VALUE1 DCD 0x200000000

VALUE2 DCD 0x00050000
    AREA DATASEG, DATA, READWRITE

RESULT DCD 0, 0       
    END
