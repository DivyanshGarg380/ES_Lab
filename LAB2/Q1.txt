/*
	Write an ARM assembly language program to copy a 16-bit variable from code memory to data memory.
*/

	AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

	LDR R0, =SRC			; Load SRC address
	LDR R1, =DST			; Load DST address
	LDR R3, [R0]			; Load data stored at address in R0
	STR R3, [R1]			; Store data in address pointed by R1

STOP 
	B STOP

SRC DCD 8
	AREA DATASEG, DATA, READWRITE

DST DCD 0
	END
