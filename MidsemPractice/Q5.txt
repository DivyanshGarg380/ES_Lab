/*
    Write an ARM program to calculate sum of N natural numbers RECURSIVELY
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =NUM
    LDR R0, [R0]

    BL SUM

    LDR R1, =RESULT 
    STR R0, [R1]

STOP    
    B STOP


SUM     
    CMP R0, #0
    BEQ BASE 

    PUSH {R0, LR}           ; save N 

    SUB R0, R0, #1
    BL SUM                  ; Recursive call on N - 1 

    MOV R1, R0              ; Contains SUM(N-1)

    POP {R0, LR}            ; Restore N 

    ADD R0, R0, R1          ; N + SUM(N-1)
    BX LR

BASE    
    MOV R0, #0
    BX LR

NUM DCD 5
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
