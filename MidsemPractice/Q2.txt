/*
    Write an ARM program to find the larger of two unsigned 32-bit numbers using a function.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =NUM1
    LDR R0, [R0]

    LDR R1, =NUM2
    LDR R1, [R1]

    BL MAX_NUM

    LDR R2, =RESULT
    STR R0, [R2]

STOP
    B STOP

MAX_NUM
    CMP R0, R1
    BHS DONE               ; R0 >= R1

    MOV R0, R1             ; if R1 is larger -> move it to R0

DONE    
    BX LR

NUM1 DCD 45
NUM2 DCD 72
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
