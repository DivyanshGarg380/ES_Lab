/*
    Write an ARM assembly program to add two 32-bit numbers using a separate function ADD_NUM.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =NUM1
    LDR R0, [R0]

    LDR R1, =NUM2
    LDR R1, [R1]

    BL ADD_NUM 

    LDR R2, =RESULT
    STR R0, [R2]

STOP
    B STOP

ADD_NUM 
    ADD R0, R0, R1
    BX LDR

NUM1 DCD 25

NUM2 DCD 15
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
