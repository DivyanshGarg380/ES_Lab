/*
    Write an ARM assembly program to count the number of positive and 
    negative numbers in an array of 10 signed 32-bit numbers.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

Reset_Handler

    LDR R0, =ARRAY        ; R0 = address of array
    MOV R1, #10           ; 10 numbers

    MOV R5, #0            ; positive count
    MOV R6, #0            ; negative count

LOOP
    LDR R2, [R0]          ; load current number

    CMP R2, #0            ; compare with zero
    BGE POSITIVE          ; if number >= 0, positive

    ADD R6, R6, #1        ; negative count++
    B NEXT

POSITIVE
    ADD R5, R5, #1        ; positive count++

NEXT
    ADD R0, R0, #4        ; move to next number
    SUB R1, R1, #1        ; decrease count

    CMP R1, #0
    BNE LOOP              ; continue

STOP
    B STOP


ARRAY
    DCD 10,-5,20,-8,-2,7,4,-1,6,9

    END
