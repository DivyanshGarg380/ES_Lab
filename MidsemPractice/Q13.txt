/*
    Write an ARM assembly program to:

    - Create a function DIVIDE to divide a 32-bit number by repeated subtraction.
    - Create a function PRIME which uses DIVIDE to check whether a number is prime.
    - Create an array of 5 32-bit numbers.
    - Use PRIME to count the number of prime numbers in the array.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R4, =ARRAY           ; Address of array
    MOV R5, #5               ; 5 elements
    MOV R6, #0               ; Prime count 

LOOP 
    LDR R0, [R4]             ; current number 
    BL PRIME                 ; R0 = 1 if prime else 0

    CMP R0, #1 
    BNE NOT_PRIME 

    ADD R6, R6, #1           ; prime++;

NOT_PRIME
    ADD R4, R4, #4
    SUBS R5, R5, #1
    BNE LOOP 
    
    LDR R7, =RESULT
    STR R6, [R7]

STOP 
    B STOP 


PRIME
    CMP R0, #2
    BLO NOT_PRIME_FUNC

    MOV R7, R0              ; dividend
    MOV R8, #2              ; divisor 

PRIME_LOOP
    CMP R8, R7 
    BHS PRIME_FOUND         ; stop at number itself 

    MOV R0, R7              ; R0 = number 
    MOV R1, R8              ; R1 = divisor 

    BL DIVIDE               ; R0 = quotient and R1 = remainder 

    CMP R1, #0
    BEQ NOT_PRIME_FUNC

    ADD R8, R8, #1
    B PRIME_LOOP


PRIME_FOUND
    MOV R0, #1
    BX LR 

NOT_PRIME_FUNC
    MOV R0, #0
    BX LR

DIVIDE 
    MOV R2, #0              ; quotient
    MOV R3, R0              ; dividend 

DIV_LOOP    
    CMP R3, R1 
    BLO DIV_DONE
    
    SUB R3, R3, R1
    ADD R2, R2, #1

    B DIV_LOOP 

DIV_DONE 
    MOV R0, R2
    MOV R1, R3
    BX LR 

ARRAY DCD 7, 10, 13, 15, 17
    AREA DAA, DATA, READWRITE 

RESULT DCD 0
    END
