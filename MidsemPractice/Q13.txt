/*
    Write an ARM assembly program to:

    - Create a function DIVIDE to divide a 32-bit number by repeated subtraction.
    - Create a function PRIME which uses DIVIDE to check whether a number is prime.
    - Create an array of 5 32-bit numbers.
    - Use PRIME to count the number of prime numbers in the array.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handlers

    LDR R4, =ARRAY           ; Address of array
    MOV R5, #5               ; 5 elements
    MOV R6, #0               ; Prime count 

LOOP 
    LDR R0, [R4]    
    
    PUSH {R4, R5, R6, LR}    ; Save array info 
    BL PRIME                 ; Check whether number is prime    
    MOV R7, R0               ; Save PRIME result
    POP {R4, R5, R6, LR} 

    CMP R7, #1
    BNE NOT_PRIME 

    ADD R6, R6, #1           ; prime count++;

NOT_PRIME
    ADD R4, R4, #4           ; move to nxt number 
    SUB R5, R5, #1

    CMP R5, #0
    BNE LOOP 

    LDR R0, =RESULT
    STR R6, [R0]            ; Store prime count 

STOP 
    B STOP 

PRIME
    CMP R0, #2
    BLO NOT_PRIME_FUNC      ; 0 and 1 are not prime 

    PUSH {R4, R5, LR}

    MOV R4, R0              ; original number 
    MOV R5, #2              ; divisor

PRIME_LOOP 
    CMP R5, R4
    BHS PRIME_FOUND         ; stop before number itself

    MOV R0, R4              ; number 
    MOV R1, R5              ; divisor 
    BL DIVIDE               ; quotient in R0 and remainder in R1

    CMP R1, #0
    BEQ NOT_PRIME_FUNC      

    ADD R5, R5, #1
    B PRIME_LOOP

PRIME_FOUND
    MOV R0, #1
    POP {R4, R5, LR}
    BX LR 

NOT_PRIME_FUNC
    MOV R0, #0
    POP {R4, R5, LR}
    BX LR 

DIVIDE 
    MOV R2, #0
    MOV R3, R0

DIV_LOOP
    CMP R3, R1
    BLO DIV_DONE
    SUB R3, R3, R1
    ADD R2, R2, #1
    B DIV_LOOP

DIV_DONE
    MOV R0, R2
    MOV R1, R3
    BX LR 


ARRAY DCD 7, 10, 13, 15, 17
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
