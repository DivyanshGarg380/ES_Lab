/*
    Write an ARM program to calculate Factorial of a number RECURSIVELY. Also make sure to check the base case properly (include 0!)
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =NUM
    LDR R0, [R0]

    BL FACT 
    LDR R1, =RESULT 
    STR R0, [R1]

STOP
    B STOP

FACT 
    CMP R0, #1
    BLE BASE            ; if N <= 1

    PUSH {R0, LR}       ; save N 

    SUB R0, R0, #1
    BL FACT 

    MOV R1, R0

    POP {R0, LR}        ; Restore original N

    MUL R0, R0, R1 
    BX LR

BASE 
    MOV R0, #1
    BX LR 

NUM DCD 5 
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
