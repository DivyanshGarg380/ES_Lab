/*
    Write an ARM Assembly language program using a user-defined function FIB to calculate the Fibonacci number of a given input.

    A 32-bit hexadecimal number is given as input.
    The program should:

    1. Extract each hexadecimal digit (nibble) from the 32-bit hexadecimal number.
    2. Pass each individual hexadecimal digit as an argument to the user-defined FIB function.
    3. The FIB function should calculate the nth Fibonacci number for each digit.
    4. Store the Fibonacci results corresponding to all the hexadecimal digits in code/read-only memory.
    5. Additionally, multiply all the hexadecimal digits together.
    6. Store the final multiplication result in a separate memory location.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =NUMBER 
    LDR R0, [R0]            ; stores the 32 bit hexa number 
    MOV R9, R0              ; copy used for extracting digits 

    LDR R3, =RESULT         ; result array address 
    MOV R4, #1              ; multiplication result 
    MOV R2, #8              ; 8 hexa digits 

LOOP 
    AND R1, R9, #0x0F       ; current hexa digit 
    MUL R4, R4, R1          ; multiplication result *= digit 

    MOV R0, R1              ; pass digit to FIB 
    BL FIB                  ; R0 = FIB(digit)

    STR R0, [R3]            ; store fiboncci result 
    ADD R3, R3, #4

    MOV R9, R9, LSR #4      ; remove current digit 

    SUBS R2, R2, #1
    BNE LOOP 

    LDR R6, =PRODUCT
    STR R4, [R6]

STOP 
    B STOP 


FIB 
    CMP R0, #0
    BEQ FIB_ZERO

    CMP R0, #1
    BEQ FIB_ONE

    PUSH {R0, LR}

    SUB R0, R0, #1
    BL FIB                  ; FIB(n-1)

    MOV R5, R0              ; R5 = FIB(n-1)

    POP {R0, LR}

    PUSH {R5, LR} 

    SUB R0, R0, #2
    BL FIB                  ; FIB(n-2)

    POP {R5, LR} 

    ADD R0, R0, R5          ; FIB(n) = FIB(n-1) + FIB(n-2)
    BX LR 

FIB_ZERO 
    MOV R0, #0
    BX LR

FIB_ONE
    MOV R0, #1
    BX LR

NUMBER DCD 0x12345678
    AREA DATA, DATA, READWRITE 

RESULT DCD 0, 0, 0, 0, 0, 0, 0, 0
PRODUCT DCD 0
    END
