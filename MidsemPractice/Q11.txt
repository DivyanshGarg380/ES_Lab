/*
    Find Largest and Smallest Number in an Array
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =ARRAY 
    LDR R5, [R0]            ; assume first element is largest 
    LDR R6, R5              ; assume first element is smallest 

    ADD R0, R0, #4          ; move to second element
    MOV R1, #9

LOOP
    LDR R2, [R0]            ; curr element

    ; check for larger value
    CMP R2, R5
    BLS CHECK_SMALL
    MOV R5, R2              ; new largest 

CHECK_SMALL 
    ; check for smaller value 
    CMP R2, R6 
    BHS NEXT
    MOV R6, R2              ; new smallest 

NEXT 
    ADD R0, R0, #4
    SUB R1, R1, #1
    CMP R1, #0
    BNE LOOP

STOP
    B STOP 

ARRAY DCD 45, 12, 78, 34, 9, 56, 23, 89, 15, 67
    END
