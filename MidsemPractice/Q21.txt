/*
    Write an ARM Assembly language program to:

    - Create a function DIVIDE which divides two 32-bit unsigned numbers using repeated subtraction.
    - Create an array of 5 32-bit numbers.
    - Using the DIVIDE function, count how many numbers in the array are exactly divisible by a given divisor.
    - Store the count in memory.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R4, =ARRAY          ; address of array 
    LDR R1, =DIVISOR 
    LDR R1, [R1]            ; divisor 

    MOV R5, #5              ; loop counter 
    MOV R6, #0              ; count

LOOP 
    LDR R7, [R4]            ; load current array element
    MOV R0, R7              ; number to divide 
    BL DIVIDE               ; divide R0 by R1 

    CMP R3, #0              ; remainder == 0 ?
    BNE NOT_DIVISIBLE

    ADD R6, R6, #1          ; count++

NOT_DIVISIBLE
    ADD R4, R4, #4          ; move to next element
    SUBS R5, R5, #1
    BNE LOOP

    LDR R7, =COUNT 
    STR R6, [R7] 

STOP 
    B STOP 

DIVIDE 
    MOV R2, #0              ; quotient
    MOV R3, R0              ; remainder = divident

DIV_LOOP 
    CMP R3, R1 
    BLO DIV_DONE            ; remainder < divisor 

    SUB R3, R3, R1 
    ADD R2, R2, #1 

    B DIV_LOOP 

DIV_DONE 
    BX LR 

ARRAY DCD 20, 35, 40, 53, 60

DIVISOR DCD 5
    AREA DATA, DATA, READWRITE 

COUNT DCD 0
    END
