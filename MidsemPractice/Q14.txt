/*
    Write an ARM Assembly program to:

    - Create a function PRIME that checks whether a number is prime.
    - Create a function FACT that calculates factorial using recursion.
    - Create an array of 5 32-bit numbers.
    - For every number in the array:
        - If it is prime, calculate its factorial.
        - Store the factorial in a result array.
        - If it is not prime, store 0.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R4, =ARRAY 
    LDR R5, =RESULT 
    MOV R6, #5

LOOP 
    LDR R0, [R4]
    
    PUSH {R4, R5, R6, LR}
    BL PRIME                ; R0 = 1 if prime, else 0
    MOV R7, R0
    POP {R4, R5, R6, LR}

    CMP R7, #1
    BNE NOT_PRIME

    LDR R0, [R4]
    BL FACT 

    STR R0, [R5]
    B NEXT 

NOT_PRIME
    MOV R0, #0 
    STR R0, [R5]

NEXT 
    ADD R4, R4, #4
    ADD R5, R5, #5
    SUB R6, R6, #1
    CMP R6, #0
    BNE LOOP 

STOP 
    B STOP

PRIME
    PUSH {R4, R5, LR}

    CMP R0, #2
    BLO NOT_PRIME_FUNC    ; 0 and 1 are not prime

    MOV R4, R0            ; R4 = number
    MOV R5, #2            ; first divisor = 2

PRIME_LOOP
    CMP R5, R4
    BHS PRIME_FOUND       ; divisor reached number

    MOV R0, R4
    MOV R1, R5
    BL DIVIDE

    CMP R1, #0            ; remainder == 0?
    BEQ NOT_PRIME_FUNC

    ADD R5, R5, #1
    B PRIME_LOOP

PRIME_FOUND
    MOV R0, #1
    POP {R4, R5, LR}
    BX LR

NOT_PRIME_FUNC
    MOV R0, #0
    POP {R4, R5, LR}
    BX LR


DIVIDE
    MOV R2, #0            ; quotient
    MOV R3, R0            ; remainder

DIV_LOOP
    CMP R3, R1
    BLO DIV_DONE

    SUB R3, R3, R1
    ADD R2, R2, #1
    B DIV_LOOP

DIV_DONE
    MOV R0, R2
    MOV R1, R3
    BX LR


FACT
    CMP R0, #1
    BLE FACT_BASE

    PUSH {R0, LR}

    SUB R0, R0, #1
    BL FACT

    MOV R1, R0
    POP {R0, LR}

    MUL R0, R0, R1
    BX LR

FACT_BASE
    MOV R0, #1
    BX LR


ARRAY DCD 5, 8, 3, 10, 7
    AREA DATA, DATA, READWRITE

RESULT DCD 0, 0, 0, 0, 0
    END
