/*
    Write an ARM program to count how many even numbers are present in an array of 10 numbers.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =ARRAY
    MOV R1, #10
    MOV R2, #0              ; even count tracker

LOOP
    LDR R3, [R0]
    AND R4, R3, #1          ; check last bit
    CMP R4, #0
    BNE NEXT                ; odd -> skip

    ADD R2, R2, #1

NEXT 
    ADD R0, R0, #4
    SUB R1, R1, #1
    CMP R1, #0
    BNE LOOP

    LDR R5, =RESULT
    STR R2, [R5]

STOP
    B STOP

ARRAY DCD 10, 15, 20, 23,30,41, 50, 61, 70, 80
    AREA DATA, DATA, READWRITE

RESULT DCD 0
    END
