/*
    Write an Assembly program to sort an array using selection sort.
*/

    AREA RESET, DATA, READONLY
 	EXPORT __Vectors
__Vectors
	DCD 0x10001000 			; stack pointer value when stack is empty
 	DCD Reset_Handler 		; reset vector

 	ALIGN
	AREA mycode, CODE, READONLY
	ENTRY
	EXPORT Reset_Handler
Reset_Handler

    LDR R0, =LIST
    MOV R1, #0              ; index i = 0

OUTER 
    CMP R1, #9              ; i < 9 ?
    BGE DONE

    MOV R2, R1              ; min = i
    ADD R3, R1, #1          ; j = i + 1

INNER
    CMP R3, #10             ; j < 10
    BGE SWAP

    LDR R4, [R0, R3, LSL #2] ; array[i]
    LDR R5, [R0, R2, LSL #2] ; array[min]

    CMP R4, R5
    BGE NEXT

    MOV R2, R3              ; min = j

NEXT
    ADD R3, R3, #1
    B INNER

SWAP    
    LDR R4, [R0, R1, LSL #2]
    LDR R5, [R0, R2, LSL #2]

    STR R5, [R0, R1, LSL #2]
    STR R4, [R0, R2, LSL #2]

    ADD R1, R1, #1
    B OUTER

DONE 
    B DONE

    AREA mydata, DATA, READWRITE

LIST DCD 45, 12, 78, 3, 25, 9, 60, 1, 34, 30
    ENDS
